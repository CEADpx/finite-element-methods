% Match the HTML theorem headings by bolding parenthetical theorem titles.
% Pandoc/Quarto assigns theorem-like environments to the three amsthm styles
% below; append the note-font override after those styles are selected.  The
% standard amsthm remark style also uses an italic heading, so override its
% heading font to keep "Remark" bold like the other environment labels.
\makeatletter
\AtBeginDocument{%
  \g@addto@macro\th@plain{\thm@notefont{\bfseries}}%
  \g@addto@macro\th@definition{\thm@notefont{\bfseries}}%
  \g@addto@macro\th@remark{\thm@headfont{\bfseries}\thm@notefont{\bfseries}}%
}
\makeatother

% Distinguish worked examples and exercises without changing theorem styles.
\definecolor{femexampleback}{HTML}{F5F9FD}
\definecolor{femexampleframe}{HTML}{8EB6D8}
\definecolor{femexerciseback}{HTML}{FFFAF0}
\definecolor{femexerciseframe}{HTML}{D4A72C}
\tcbset{
  femexamplebox/.style={
    enhanced,
    breakable,
    colback=femexampleback,
    colframe=femexampleframe,
    boxrule=0.45pt,
    leftrule=2.2pt,
    arc=1.5pt,
    left=7pt,
    right=7pt,
    top=6pt,
    bottom=6pt,
    before skip=0.9em,
    after skip=0.9em
  },
  femexercisebox/.style={
    enhanced,
    breakable,
    colback=femexerciseback,
    colframe=femexerciseframe,
    boxrule=0.45pt,
    leftrule=2.2pt,
    arc=1.5pt,
    left=7pt,
    right=7pt,
    top=6pt,
    bottom=6pt,
    before skip=0.9em,
    after skip=0.9em
  }
}

% Quarto defines these theorem environments after processing header includes.
\AtBeginDocument{%
  \tcolorboxenvironment{example}{femexamplebox}%
  \tcolorboxenvironment{exercise}{femexercisebox}%
}
