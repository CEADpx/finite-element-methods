\section{Review}

\begin{frame}{Review: diffusion problem and variational form}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.43\textwidth}
    With homogeneous Dirichlet data, the strong problem is
    \[
      \begin{aligned}
      -\divg(k\nabla u)&=f &&\text{in }\Om,\\
      u&=0 &&\text{on }\GamD,\\
      -k\nabla u\cdot\bm n&=q &&\text{on }\GamN.
      \end{aligned}
    \]
    The test and trial space is
    \[
      \Vzero=H^1_{0,\GamD}(\Om).
    \]
  \end{column}
  \begin{column}{0.54\textwidth}
    Define the bilinear and linear forms
    \[
      a(w,v)=\int_\Om k\nabla w\cdot\nabla v\,\dd x,
    \]
    \[
      \ell(v)=\int_\Om fv\,\dd x
      -\int_{\GamN}qv\,\dd s.
    \]
    The variational problem is
    \[
      \boxed{\text{find }u\in\Vzero:
      \quad a(u,v)=\ell(v)
      \quad\forall v\in\Vzero.}
    \]
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Review: finite element problem and stability estimates}
Let $\Vhzero\subset\Vzero$ be the continuous Lagrange finite element space of
polynomial degree $p\geq1$. The finite element problem is
\[
  \boxed{\text{find }u_h\in\Vhzero:
  \quad a(u_h,v_h)=\ell(v_h)
  \quad\forall v_h\in\Vhzero.}
\]
If $0<k_0\leq k(\bx)\leq k_1$, then
\[
  |a(w,v)|\leq
  M\norm[H^1(\Om)]{w}\norm[H^1(\Om)]{v},
  \qquad M=k_1,
\]
\[
  a(v,v)\geq
  \alpha\norm[H^1(\Om)]{v}^2,
  \qquad \alpha=\frac{k_0}{1+C_P^2}.
\]
Here $M$ is the continuity constant and $\alpha$ is the coercivity constant.
Since $a$ is symmetric,
\[
  \norm[a]{v}=\sqrt{a(v,v)}
\]
defines the energy norm.
\end{frame}

\begin{frame}{Review: orthogonality, C\'ea's lemma, and interpolation}
Let $e=u-u_h$. For any $v_h\in\Vhzero$,
\[
  a(e,v_h)=a(u,v_h)-a(u_h,v_h)
  =\ell(v_h)-\ell(v_h)=0.
\]
This is Galerkin orthogonality. C\'ea's lemma gives
\[
  \norm[H^1(\Om)]{e}
  \leq\frac{M}{\alpha}
  \inf_{w_h\in\Vhzero}\norm[H^1(\Om)]{u-w_h}.
\]
For degree-$p$ interpolation on a shape-regular mesh, if
$u\in H^s(\Om)$ and $2\leq s\leq p+1$,
\[
  \norm[L^2(\Om)]{u-\Ih u}
  \leq C_I\,h^s\seminorm[H^s(\Om)]{u},
\]
\[
  \seminorm[H^1(\Om)]{u-\Ih u}
  \leq C_I\,h^{s-1}\seminorm[H^s(\Om)]{u}.
\]
\end{frame}

\begin{frame}{Review: a priori estimates obtained previously}
Combining best approximation with the interpolation estimate gives
\[
  \norm[a]{e}
  \leq C_I\sqrt{k_1}\,h^{s-1}\seminorm[H^s(\Om)]{u}.
\]
The coefficient bounds give the $H^1$-seminorm estimate
\[
  \boxed{\seminorm[H^1(\Om)]{e}
  \leq C_I\sqrt{\frac{k_1}{k_0}}\,
  h^{s-1}\seminorm[H^s(\Om)]{u}.}
\]
Using Poincar\'e's inequality gives
\[
  \norm[H^1(\Om)]{e}
  \leq C_I\sqrt{\frac{k_1(1+C_P^2)}{k_0}}\,
  h^{s-1}\seminorm[H^s(\Om)]{u}.
\]
Therefore, the direct $L^2$ estimate is only
\[
  \norm[L^2(\Om)]{e}
  \leq\norm[H^1(\Om)]{e}
  \leq C h^{s-1}\seminorm[H^s(\Om)]{u}.
\]
\end{frame}

\section{Predicted rates}

\begin{frame}[plain]
  \centering\vfill
  {\LARGE\bfseries What rates should we expect?\par}
  \vspace{0.45cm}
  {\large Polynomial degree and solution regularity both limit the answer.\par}
  \vfill
\end{frame}

\begin{frame}{Predicted rates for a smooth solution}
Assume $u\in H^{p+1}(\Om)$. The $L^2$ column also requires the dual-regularity
assumption introduced next.
\par
\vspace{0.2cm}
\centering
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{c|cc}
  \toprule
  degree $p$ & energy and $H^1$ error & $L^2$ error\\
  \midrule
  $1$ & $O(h)$ & $O(h^2)$\\
  $2$ & $O(h^2)$ & $O(h^3)$\\
  $3$ & $O(h^3)$ & $O(h^4)$\\
  \bottomrule
\end{tabular}
\par
\vspace{0.4cm}
$E(h)=O(h^r)$ means $E(h)\leq Ch^r$, where $C$ is independent of $h$.
The exponent $r$ is the convergence rate in the stated norm.
\end{frame}

\section{The improved \texorpdfstring{$L^2$}{L2} estimate}

\begin{frame}[plain]
  \centering\vfill
  {\LARGE\bfseries Why can the $L^2$ error converge faster?\par}
  \vspace{0.45cm}
  {\large A dual problem supplies one additional factor of $h$.\par}
  \vfill
\end{frame}

\begin{frame}{The dual problem associated with the error}
Let $e=u-u_h$. Using the same diffusion bilinear form
\[
  a(w,v)=\int_\Om k\nabla w\cdot\nabla v\,\dd x,
\]
the \textbf{dual problem} is to find $z\in\Vzero$ such that
\[
  a(v,z)=(e,v)_{L^2(\Om)}
  \qquad\forall v\in\Vzero.
\]
Because $a$ is symmetric, this is the variational form of
\[
  \begin{aligned}
  -\divg(k\nabla z)&=e &&\text{in }\Om,\\
  z&=0 &&\text{on }\GamD,\\
  k\nabla z\cdot\bm n&=0 &&\text{on }\GamN.
  \end{aligned}
\]
We assume that $z\in H^2(\Om)$ and
\[
  \seminorm[H^2(\Om)]{z}
  \leq C_{\mathrm{reg}}\norm[L^2(\Om)]{e}.
\]
\end{frame}

\begin{frame}{The error identity}
Choose $v=e$ in the dual problem:
\[
  \norm[L^2(\Om)]{e}^{2}=a(e,z).
\]
The $H^2$ regularity of $z$ ensures that its interpolant is defined, and
$\Ih z\in\Vhzero$. Galerkin orthogonality states
\[
  a(e,v_h)=0\qquad\forall v_h\in\Vhzero.
\]
Use the particular test function $v_h=\Ih z$:
\[
  a(e,\Ih z)=0.
\]
By bilinearity in the second argument,
\[
  \boxed{\norm[L^2(\Om)]{e}^{2}
  =a(e,z)-a(e,\Ih z)
  =a(e,z-\Ih z).}
\]
The right-hand side now contains the interpolation error of the dual
solution.
\end{frame}

\begin{frame}{Obtain the additional factor of \texorpdfstring{$h$}{h}}
For the diffusion form, the coefficient bound and Cauchy--Schwarz inequality
give
\[
  |a(w,v)|
  \leq k_1\seminorm[H^1(\Om)]{w}
  \seminorm[H^1(\Om)]{v}.
\]
Use this continuity estimate, interpolation with $s=2$, and dual regularity:
\begin{align*}
  \norm[L^2(\Om)]{e}^{2}
  &=a(e,z-\Ih z)\\
  &\leq k_1\seminorm[H^1(\Om)]{e}
       \seminorm[H^1(\Om)]{z-\Ih z}\\
  &\leq k_1C_I\,h\seminorm[H^1(\Om)]{e}
       \seminorm[H^2(\Om)]{z}\\
  &\leq k_1C_IC_{\mathrm{reg}}h\seminorm[H^1(\Om)]{e}
       \norm[L^2(\Om)]{e}.
\end{align*}
\end{frame}

\begin{frame}{Combine with the \texorpdfstring{$H^1$}{H1} estimate}
If $\norm[L^2(\Om)]{e}\ne0$, cancel one error factor to obtain
\[
  \norm[L^2(\Om)]{e}
  \leq k_1C_IC_{\mathrm{reg}}h
  \seminorm[H^1(\Om)]{e}.
\]
The recalled $H^1$-seminorm estimate is
\[
  \seminorm[H^1(\Om)]{e}
  \leq C_I\sqrt{\frac{k_1}{k_0}}\,
  h^{s-1}\seminorm[H^s(\Om)]{u}.
\]
Combining the two estimates gives
\begin{align*}
  \norm[L^2(\Om)]{e}
  &\leq C_I^2C_{\mathrm{reg}}k_1
  \sqrt{\frac{k_1}{k_0}}\,
  h^s\seminorm[H^s(\Om)]{u}.
\end{align*}
For degree-$p$ elements, provided the exact solution belongs to
$H^{p+1}(\Om)$ and the dual problem is $H^2$-regular, choose $s=p+1$:
\[
  \boxed{\norm[L^2(\Om)]{u-u_h}\leq
  C h^{p+1}\seminorm[H^{p+1}(\Om)]{u}.}
\]
\end{frame}

\section{Numerical verification}

\begin{frame}[plain]
  \centering\vfill
  {\LARGE\bfseries Numerical verification\par}
  \vspace{0.45cm}
  {\large Test the predicted rates for a smooth elasticity problem.\par}
  \vfill
\end{frame}

\begin{frame}{Observed convergence rate}
If $E(h)\approx Ch^r$, two consecutive meshes give
\[
  \boxed{r_{\mathrm{obs}}
  =\frac{\log(E(h_1)/E(h_2))}{\log(h_1/h_2)}.}
\]
For uniform refinement with $h_2=h_1/2$,
\[
  r_{\mathrm{obs}}=\log_2\!\left(\frac{E(h)}{E(h/2)}\right).
\]
Therefore:
\[
\begin{array}{c|ccc}
  \text{error reduction after halving }h & 2 & 4 & 8\\ \hline
  r_{\mathrm{obs}} & 1 & 2 & 3
\end{array}
\]
We look for stabilization of $r_{\mathrm{obs}}$ over several refinements,
not agreement from a single pair of meshes.
\end{frame}

\begin{frame}{What a verification study requires}
To compare a theorem with a computation, we need:
\begin{enumerate}
  \item an exact solution with the regularity assumed by the theorem;
  \item data chosen so that this exact solution satisfies the boundary-value
  problem;
  \item a mesh sequence that preserves element quality;
  \item the error evaluated in each norm for which a rate is predicted;
  \item sufficiently accurate quadrature for the error integrals; and
  \item enough refinements to reach the asymptotic range.
\end{enumerate}
\vspace{0.25cm}

An attractive solution plot is not a convergence test. Nodal errors alone are
also insufficient because the error between nodes may be nonzero.
\end{frame}

\begin{frame}{Linear elasticity problem used in the notebook}
For plane strain, with $\lambda=\mu=1$,
\[
  \beps(\bu)=\tfrac12\left(\nabla\bu+\nabla\bu^T\right),
  \qquad
  \bsig(\bu)=2\mu\beps(\bu)
  +\lambda\operatorname{tr}(\beps(\bu))\bm I.
\]
The finite element problem is
\[
  \int_\Om \bsig(\bu_h):\beps(\bv_h)\dd x
  =\int_\Om \bm b\cdot\bv_h\dd x
  \qquad\forall\bv_h\in\mathcal V_{h,0}.
\]
For a smooth exact displacement and $P_1$ triangles, the predictions are
\[
  \norm[L^2(\Om;\R^2)]{\bu-\bu_h}=O(h^2),
\]
\[
  \seminorm[H^1(\Om;\R^2)]{\bu-\bu_h}=O(h),
  \qquad
  \norm[a_{\mathrm{el}}]{\bu-\bu_h}=O(h).
\]
\end{frame}

\begin{frame}{Manufactured smooth displacement}
Let $\Om=(0,1)^2$ and prescribe $\bu=\bm 0$ on $\partial\Om$. Choose
\[
  \bu_\star(x,y)=
  \begin{bmatrix}
    \sin(\pi x)\sin(\pi y)\\
    \sin(2\pi x)\sin(\pi y)
  \end{bmatrix}.
\]
The body force is then defined by
\[
  \bm b=-\divg\bsig(\bu_\star).
\]
This construction guarantees that $\bu_\star$ solves the strong problem. It
is smooth and is not contained in the $P_1$ finite element space.
\vspace{0.2cm}

We can therefore compute the $L^2$, $H^1$-seminorm, and energy errors directly
and compare their observed rates with the predicted rates.
\end{frame}

\begin{frame}{Notebook workflow}
For each subdivision count $n=4,8,16,32$:
\begin{enumerate}
  \item construct a triangular mesh of the unit square;
  \item define the vector-valued $P_1$ space;
  \item form $\bm b=-\divg\bsig(\bu_\star)$;
  \item assemble and solve the elasticity problem;
  \item integrate the three error norms; and
  \item compute the observed rates between successive meshes.
\end{enumerate}
\vspace{0.3cm}

The notebook follows this sequence explicitly. Plotting helpers are kept
separate so that the finite element algorithm remains visible during the
demonstration.
\end{frame}

    \[

    \]

\begin{frame}{The corner limits the observed convergence rate}
For linear elements, the estimate derived earlier gives
\[
  \norm[a_{\mathrm{el}}]{\bu-\bu_h}
  \leq C h\seminorm[H^2(\Om)]{\bu}
\]
provided the exact displacement belongs to $[H^2(\Om)]^2$.
\vspace{0.2cm}

The successive-mesh energy differences, rather than exact errors, decrease and
provide numerical evidence of global energy convergence. Their observed rates,
approximately $0.65$ and $0.69$, are nevertheless below the smooth-solution
rate $1$. This is consistent with the reentrant corner reducing the regularity
of the exact displacement.
\vspace{0.2cm}

At the same time, the largest elementwise strain-energy density increases near the
corner. Global energy convergence therefore does not prevent a local strain and
stress concentration. Chapter 8 uses residual indicators to identify where
local refinement is needed.
\end{frame}

\begin{frame}{Summary}
\begin{itemize}
  \item C\'ea's lemma and interpolation give energy-, $H^1$-seminorm, and
  $H^1$-norm estimates proportional to $h^{s-1}$.
  \item If the dual problem is $H^2$-regular, Galerkin orthogonality introduces
  the interpolation error $z-\Ih z$ and supplies one additional power of $h$
  in the $L^2$ estimate.
  \item The smooth $P_1$ elasticity example exhibits rates $2$ in $L^2$ and
  $1$ in the $H^1$ seminorm and energy norm.
  \item For the L-shaped example, successive-mesh energy differences converge
  at rates near $0.7$, below the smooth $P_1$ energy rate $1$.
  \item The reentrant corner reduces regularity and produces a local
  strain-energy concentration, motivating local refinement.
\end{itemize}
\end{frame}
